    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]

    # Add the detector's own train-split boxes (labelled by overlap with a GT crop) so stage 2 is
    # trained on the box distribution it must classify at test time, not only perfect GT boxes.
    tr = [d for d in dets if d["split"] == "train"]
    by_img = {}
    for r in crops:
        by_img.setdefault(r["image"], []).append(r)

    def iou(a, b):
        ix1, iy1 = max(a["x1"], b["x1"]), max(a["y1"], b["y1"])
        ix2, iy2 = min(a["x2"], b["x2"]), min(a["y2"], b["y2"])
        iw, ih = max(0.0, ix2 - ix1), max(0.0, iy2 - iy1)
        inter = iw * ih
        ua = (a["x2"] - a["x1"]) * (a["y2"] - a["y1"]) + (b["x2"] - b["x1"]) * (b["y2"] - b["y1"]) - inter
        return inter / ua if ua > 0 else 0.0

    extra = []
    for d in tr:
        best, best_i = None, 0.0
        for r in by_img.get(d["image"], ()):
            v = iou(d, r)
            if v > best_i:
                best, best_i = r, v
        if best is not None and best_i > 0.5:
            nd = dict(d)
            nd["cls"] = best["cls"]
            extra.append(nd)
    crops = crops + extra
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
