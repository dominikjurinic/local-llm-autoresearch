    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]

    # Extra training boxes: stage-1 detector boxes on train images that overlap a labelled crop (IoU > 0.5),
    # labelled with that crop's class. These match the distribution of the eval boxes we must classify.
    by_img = {}
    for r in crops:
        by_img.setdefault(r["image"], []).append(r)
    extra = []
    for d in dets:
        if d["split"] != "train":
            continue
        rows = by_img.get(d["image"])
        if not rows:
            continue
        best = None
        for r in rows:
            ix1, iy1 = max(d["x1"], r["x1"]), max(d["y1"], r["y1"])
            ix2, iy2 = min(d["x2"], r["x2"]), min(d["y2"], r["y2"])
            inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
            area = (d["x2"] - d["x1"]) * (d["y2"] - d["y1"]) + (r["x2"] - r["x1"]) * (r["y2"] - r["y1"]) - inter
            if area <= 0:
                continue
            iou = inter / area
            if best is None or iou > best[0]:
                best = (iou, r)
        if best is not None and best[0] > 0.5:
            e = dict(d)
            e["cls"] = best[1]["cls"]
            extra.append(e)
    crops = crops + extra
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
    images = {p: np.asarray(Image.open(os.path.join(args.data, "images", p)).convert("RGB")) for p in needed}
    print(f"train {len(crops)} boxes (incl. {len(extra)} detector boxes), eval {len(ev)} detections, {len(images)} images ({time.time() - T0:.0f}s)", flush=True)
