    h, w = img.shape[:2]
    cx, cy = (r["x1"] + r["x2"]) / 2, (r["y1"] + r["y2"]) / 2
    side = max(max(r["x2"] - r["x1"], r["y2"] - r["y1"]) * context, MIN_CROP)
    if train:  # light geometric jitter
        cx += random.uniform(-0.1, 0.1) * side
        cy += random.uniform(-0.1, 0.1) * side
        side *= random.uniform(0.85, 1.2)
    s = max(2, int(round(side)))
    x0, y0 = int(round(cx - s / 2)), int(round(cy - s / 2))

    sx0, sy0, sx1, sy1 = max(0, x0), max(0, y0), min(w, x0 + s), min(h, y0 + s)
    if sx1 > sx0 and sy1 > sy0:
        canvas[sy0 - y0:sy1 - y0, sx0 - x0:sx1 - x0] = img[sy0:sy1, sx0:sx1]
    return np.asarray(Image.fromarray(canvas).resize((IMG_SIZE, IMG_SIZE), Image.BILINEAR), dtype=np.float32) / 255.0


def paste_pore(img, r):
    """Copy-paste: paste the pore's tight crop onto a random spot of a random other image."""
    H, W = img.shape[:2]
    src = make_crop(img, r, TIGHT_CONTEXT, train=False)  # tight crop of the pore, 160x160
    scale = random.uniform(0.7, 1.3)
    ps = max(4, int(round(160 * scale)))
    tile = np.asarray(Image.fromarray((src * 255.0).astype(np.uint8)).resize((ps, ps), Image.BILINEAR),
                      dtype=np.uint8)
    out = img.copy()
    px = random.randint(0, max(0, W - ps))
    py = random.randint(0, max(0, H - ps))
    out[py:py + ps, px:px + ps] = tile
    return out, (px + ps / 2, py + ps / 2, ps)

class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, images, train):
        self.rows, self.images, self.train = rows, images, train
        self.img_names = list(images)

    def __len__(self):
        return len(self.rows)

    def __getitem__(self, i):
        r = self.rows[i]
        img = self.images[r["image"]]
        if self.train and r.get("cls") == 0 and random.random() < 0.5:
            # copy-paste pore onto a random other image; crop the box from the modified image
            base = self.images[random.choice(self.img_names)]
            img, (cx, cy, side) = paste_pore(base, r)
            r = dict(r, x1=cx - side / 2, y1=cy - side / 2, x2=cx + side / 2, y2=cy + side / 2)
