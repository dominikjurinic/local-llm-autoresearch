def make_crop(img, r, context, train, edge=False):
    h, w = img.shape[:2]
    cx, cy = (r["x1"] + r["x2"]) / 2, (r["y1"] + r["y2"]) / 2
    side = max(max(r["x2"] - r["x1"], r["y2"] - r["y1"]) * context, MIN_CROP)
    if train:  # light geometric jitter
        cx += random.uniform(-0.1, 0.1) * side
        cy += random.uniform(-0.1, 0.1) * side
        side *= random.uniform(0.85, 1.2)
    s = max(2, int(round(side)))
    x0, y0 = int(round(cx - s / 2)), int(round(cy - s / 2))
    canvas = np.full((s, s, 3), 114, dtype=np.uint8)  # gray padding outside the image
    sx0, sy0, sx1, sy1 = max(0, x0), max(0, y0), min(w, x0 + s), min(h, y0 + s)
    if sx1 > sx0 and sy1 > sy0:
        canvas[sy0 - y0:sy1 - y0, sx0 - x0:sx1 - x0] = img[sy0:sy1, sx0:sx1]
    crop = np.asarray(Image.fromarray(canvas).resize((IMG_SIZE, IMG_SIZE), Image.BILINEAR), dtype=np.float32) / 255.0
    if edge:
        g = crop.mean(axis=2)
        kx = np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]], dtype=np.float32)
        ky = kx.T
        gx = np.zeros_like(g); gy = np.zeros_like(g)
        for i in range(1, IMG_SIZE - 1):
            for j in range(1, IMG_SIZE - 1):
                patch = g[i - 1:i + 2, j - 1:j + 2]
                gx[i, j] = float((patch * kx).sum())
                gy[i, j] = float((patch * ky).sum())
        mag = np.sqrt(gx ** 2 + gy ** 2)
        m = mag.max()
        if m > 1e-6:
            mag = mag / m
        crop = np.dstack([mag, mag, mag])
    return crop
