    def loss_fn(logits, y):
        ce = crit(logits, y)
        margin = logits.new_zeros(())
        for c in range(4):
            sel = (y == c)
            if not sel.any():
                continue
            lc = logits[sel]
            for d in CONFUSED[c]:
                margin = torch.max(margin, (lc[:, d] - lc[:, c] + MARGIN).clamp(min=0.0).mean())
        return ce + margin
