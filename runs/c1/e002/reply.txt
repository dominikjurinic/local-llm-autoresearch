def box_size_feats(r):
    """Log long-side and log aspect ratio of the box, as two constant channels."""
    bw = max(r["x2"] - r["x1"], 1e-6)
    bh = max(r["y2"] - r["y1"], 1e-6)
    return float(np.log(max(bw, bh) / 64.0)), float(np.log(bw / bh))
