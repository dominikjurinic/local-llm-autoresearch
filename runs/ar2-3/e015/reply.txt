    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]
    # Label train detector boxes by IoU>0.5 overlap with GT crops, add to training set
    gt_by_image = {}
    for c in crops:
        gt_by_image.setdefault(c["image"], []).append(c)
    det_labeled = []
    for d in [x for x in dets if x["split"] == "train"]:
        img = d["image"]
        if img not in gt_by_image:
            continue
        dx1, dy1, dx2, dy2 = d["x1"], d["y1"], d["x2"], d["y2"]
        best_iou, best_cls = 0.0, -1
        for c in gt_by_image[img]:
            ix1, iy1 = max(dx1, c["x1"]), max(dy1, c["y1"])
            ix2, iy2 = min(dx2, c["x2"]), min(dy2, c["y2"])
            inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
            union = (dx2 - dx1) * (dy2 - dy1) + (c["x2"] - c["x1"]) * (c["y2"] - c["y1"]) - inter
            iou = inter / max(union, 1e-6)
            if iou > best_iou:
                best_iou, best_cls = iou, c["cls"]
        if best_iou > 0.5:
            det_labeled.append({"image": img, "x1": dx1, "y1": dy1, "x2": dx2, "y2": dy2, "cls": best_cls})
    train_rows = list(crops) + det_labeled
    needed = {r["image"] for r in train_rows} | {d["image"] for d in ev}
    images = {p: np.asarray(Image.open(os.path.join(args.data, "images", p)).convert("RGB")) for p in needed}
    print(f"train {len(train_rows)} boxes ({len(crops)} GT + {len(det_labeled)} det), eval {len(ev)} detections, {len(images)} images ({time.time() - T0:.0f}s)", flush=True)
