```python
def match_det_conf(crops, dets_train):
    """For each GT crop, find the best-matching detector box and return its confidence."""
    # Group detector boxes by image
    det_by_img = {}
    for d in dets_train:
        det_by_img.setdefault(d["image"], []).append(d)
    
    weights = []
    for c in crops:
        img = c["image"]
        if img not in det_by_img:
            weights.append(0.5)
            continue
        # Compute IoU with all detector boxes on this image
        best_iou = 0
        best_conf = 0.5
        cx1, cy1, cx2, cy2 = c["x1"], c["y1"], c["x2"], c["y2"]
        for d in det_by_img[img]:
            dx1, dy1, dx2, dy2 = d["x1"], d["y1"], d["x2"], d["y2"]
            ix1, iy1 = max(cx1, dx1), max(cy1, dy1)
            ix2, iy2 = min(cx2, dx2), min(cy2, dy2)
            inter = max(0, ix2 - ix1) * max(0, iy2 - iy1)
            area_c = (cx2 - cx1) * (cy2 - cy1)
            area_d = (dx2 - dx1) * (dy2 - dy1)
            union = area_c + area_d - inter
            iou = inter / union if union > 0 else 0
            if iou > best_iou:
                best_iou = iou
                best_conf = d["conf"]
        # Use confidence as weight, with a floor
        w = max(0.1, best_conf) if best_iou > 0.3 else 0.5
        weights.append(w)
    return weights
```

def match_det_conf(crops, dets_train):
    """For each GT crop, find the best-matching detector box and return its confidence as a weight."""
    det_by_img = {}
    for d in dets_train:
        det_by_img.setdefault(d["image"], []).append(d)
    weights = []
    for c in crops:
        img = c["image"]
        if img not in det_by_img:
            weights.append(0.5)
            continue
        cx1, cy1, cx2, cy2 = c["x1"], c["y1"], c["x2"], c["y2"]
        area_c = max(1e-6, (cx2 - cx1) * (cy2 - cy1))
        best_iou, best_conf = 0.0, 0.5
        for d in det_by_img[img]:
            dx1, dy1, dx2, dy2 = d["x1"], d["y1"], d["x2"], d["y2"]
            ix1, iy1 = max(cx1, dx1), max(cy1, dy1)
            ix2, iy2 = min(cx2, dx2), min(cy2, dy2)
            inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
            area_d = max(1e-6, (dx2 - dx1) * (dy2 - dy1))
            iou = inter / (area_c + area_d - inter)
            if iou > best_iou:
                best_iou, best_conf = iou, d["conf"]
        weights.append(max(0.1, best_conf) if best_iou > 0.3 else 0.5)
    return weights
