    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]
    all_crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    crops_by_img = {}
    for c in all_crops:
        crops_by_img.setdefault(c["image"], []).append(c)
    train_rows = []
    for d in [x for x in dets if x["split"] == "train"]:
        best_cls, best_iou = -1, 0.0
        for c in crops_by_img.get(d["image"], []):
            ix1, iy1 = max(d["x1"], c["x1"]), max(d["y1"], c["y1"])
            ix2, iy2 = min(d["x2"], c["x2"]), min(d["y2"], c["y2"])
            inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
            union = (d["x2"]-d["x1"])*(d["y2"]-d["y1"]) + (c["x2"]-c["x1"])*(c["y2"]-c["y1"]) - inter
            iou = inter / max(union, 1e-6)
            if iou > best_iou:
                best_iou, best_cls = iou, c["cls"]
        if best_iou > 0.5:
            d["cls"] = best_cls
            train_rows.append(d)
    crops = train_rows
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
