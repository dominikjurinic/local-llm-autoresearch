def mixup(xt, xw, y, alpha=0.2):
    lam = torch.distributions.Beta(alpha, alpha).sample().item()
    idx = torch.randperm(len(y))
    xt = lam * xt + (1 - lam) * xt[idx]
    xw = lam * xw + (1 - lam) * xw[idx]
    y1, y2 = y, y[idx]
    return xt, xw, y1, y2, lam
