    def __getitem__(self, i):
        r = self.rows[i]
        img = self.images[r["image"]]
        fh = fv = False
        bright = 1.0
        ch_j = np.ones(3, dtype=np.float32)
        if self.train:  # same flips, brightness and colour jitter for both views
            fh = random.random() < 0.5
            fv = random.random() < 0.5
            bright = random.uniform(0.8, 1.2)
            ch_j = np.array([random.uniform(0.92, 1.08) for _ in range(3)], dtype=np.float32)
        views = []
        for ctx in (TIGHT_CONTEXT, CONTEXT):
            x = make_crop(img, r, ctx, self.train)
            if fh:
                x = x[:, ::-1]
            if fv:
                x = x[::-1, :]
            if self.train:
                x = np.clip(x * bright, 0, 1)
                x = np.clip(x * ch_j, 0, 1)
            x = (x - MEAN) / STD
            views.append(torch.from_numpy(np.ascontiguousarray(x.transpose(2, 0, 1))))
        return views[0], views[1], r.get("cls", -1)
