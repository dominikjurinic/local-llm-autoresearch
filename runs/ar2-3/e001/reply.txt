IOU_THRESH = 0.5        # min IoU with a GT crop to label a det box as that class


def compute_iou_norm(d, c):
    dx1, dy1 = d["x1"] / d["img_w"], d["y1"] / d["img_h"]
    dx2, dy2 = d["x2"] / d["img_w"], d["y2"] / d["img_h"]
    cx1, cy1 = c["x1"] / c["img_w"], c["y1"] / c["img_h"]
    cx2, cy2 = c["x2"] / c["img_w"], c["y2"] / c["img_h"]
    ix1, iy1 = max(dx1, cx1), max(dy1, cy1)
    ix2, iy2 = min(dx2, cx2), min(dy2, cy2)
    inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
    ad = (dx2 - dx1) * (dy2 - dy1)
    ac = (cx2 - cx1) * (cy2 - cy1)
    union = ad + ac - inter
    return inter / union if union > 0 else 0.0


def label_det_boxes(dets_train, crops):
    from collections import defaultdict
    crops_by_img = defaultdict(list)
    for c in crops:
        crops_by_img[c["image"]].append(c)
    labels = []
    for d in dets_train:
        best_iou, best_cls = 0.0, 4
        for c in crops_by_img.get(d["image"], []):
            iou = compute_iou_norm(d, c)
            if iou > best_iou:
                best_iou, best_cls = iou, c["cls"]
        labels.append(best_cls if best_iou > IOU_THRESH else 4)
    return labels
