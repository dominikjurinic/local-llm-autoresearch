```python
def random_erase(x, scale=(0.05, 0.25)):
    """Randomly erase a rectangular region of the image."""
    h, w = x.shape[:2]
    area = h * w
    for _ in range(2):  # erase 2 regions
        erase_area = random.uniform(*scale) * area
        eh = int(math.sqrt(erase_area * random.uniform(0.5, 2.0)))
        ew = int(erase_area / max(eh, 1))
        eh = min(eh, h)

            y = random.randint(0, h - eh)
            x0 = random.randint(0, w - ew)
            x[y:y+eh, x0:x0+ew] = random.random()
    return x
```
