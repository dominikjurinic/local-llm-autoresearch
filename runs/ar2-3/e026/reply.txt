def random_erase(x, n=2, scale=(0.05, 0.25)):
    h, w = x.shape[:2]
    area = h * w
    for _ in range(n):
        erase_area = random.uniform(*scale) * area
        ratio = random.uniform(0.5, 2.0)
        eh = max(1, min(h, int(math.sqrt(erase_area * ratio))))
        ew = max(1, min(w, int(erase_area / max(eh, 1))))
        y = random.randint(0, h - eh)
        x0 = random.randint(0, w - ew)
        x[y:y + eh, x0:x0 + ew] = random.random()
    return x
