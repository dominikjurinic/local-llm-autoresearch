    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]
    # Match train detector boxes to true crops by IoU; add as extra training examples
    tr_dets = [d for d in dets if d["split"] == "train"]
    crops_by_img = {}
    for c in crops:
        crops_by_img.setdefault(c["image"], []).append(c)
    extra = []
    for d in tr_dets:
        cands = crops_by_img.get(d["image"], [])
        best_iou, best_cls = 0.0, -1
        for c in cands:
            x1 = max(d["x1"], c["x1"]); y1 = max(d["y1"], c["y1"])
            x2 = min(d["x2"], c["x2"]); y2 = min(d["y2"], c["y2"])
            inter = max(0, x2 - x1) * max(0, y2 - y1)
            union = (d["x2"]-d["x1"])*(d["y2"]-d["y1"]) + (c["x2"]-c["x1"])*(c["y2"]-c["y1"]) - inter
            iou = inter / max(union, 1e-6)
            if iou > best_iou:
                best_iou, best_cls = iou, c["cls"]
        if best_iou > 0.3 and best_cls >= 0:
            extra.append({**d, "cls": best_cls})
    crops = crops + extra
