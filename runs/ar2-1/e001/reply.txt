    # Label dets_train by IoU with crops (background = class 4 if no match)
    crops_by_img = {}
    for c in crops_all:
        crops_by_img.setdefault(c["image"], []).append(c)

    def _iou(r1, r2):
        x1 = max(r1["x1"], r2["x1"]); y1 = max(r1["y1"], r2["y1"])
        x2 = min(r1["x2"], r2["x2"]); y2 = min(r1["y2"], r2["y2"])
        inter = max(0.0, x2 - x1) * max(0.0, y2 - y1)
        a1 = (r1["x2"] - r1["x1"]) * (r1["y2"] - r1["y1"])
        a2 = (r2["x2"] - r2["x1"]) * (r2["y2"] - r2["y1"])
        return inter / max(1e-6, a1 + a2 - inter)

    train_rows = []
    for d in dets_train:
        best_iou, best_cls = 0.0, 4
        for c in crops_by_img.get(d["image"], []):
            i = _iou(d, c)
            if i > best_iou:
                best_iou, best_cls = i, c["cls"]
        d["cls"] = best_cls if best_iou >= 0.5 else 4
        train_rows.append(d)
