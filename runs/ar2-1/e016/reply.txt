def random_erasing(x, p=0.3, scale=(0.02, 0.15)):
    """x is HxWx3 float array in [0,1]. Randomly masks a small rectangle."""
    if random.random() > p:
        return x
    h, w = x.shape[:2]
    area = h * w
    for _ in range(10):
        target_area = random.uniform(*scale) * area
        aspect = random.uniform(0.3, 3.3)
        eh = int(math.sqrt(target_area * aspect))
        ew = int(math.sqrt(target_area / aspect))
        if 0 < eh < h and 0 < ew < w:
            y = random.randint(0, h - eh)
            x0 = random.randint(0, w - ew)
            x[y:y + eh, x0:x0 + ew] = random.uniform(0, 1)
            return x
    return x
