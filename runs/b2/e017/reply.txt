    # Background training boxes: detector boxes on training images that overlap no GT defect.

    bg = []
    for d in dets:
        if d["split"] != "train":

        hit = False
        for gg in by_img.get(d["image"], ()):
            ix1, iy1 = max(d["x1"], gg["x1"]), max(d["y1"], gg["y1"])
            ix2, iy2 = min(d["x2"], gg["x2"]), min(d["y2"], gg["y2"])
            inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
            uni = (d["x2"] - d["x1"]) * (d["y2"] - d["y1"]) + (gg["x2"] - gg["x1"]) * (gg["y2"] - gg["y1"]) - inter
