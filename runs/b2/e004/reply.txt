    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]
    # Label the detector's own train boxes by IoU>0.5 overlap with the ground-truth crops,
    # so the classifier is trained on the same (slightly misplaced) box distribution it sees at eval time.
    def iou(a, b):
        ix1, iy1 = max(a["x1"], b["x1"]), max(a["y1"], b["y1"])
        ix2, iy2 = min(a["x2"], b["x2"]), min(a["y2"], b["y2"])
        iw, ih = max(0.0, ix2 - ix1), max(0.0, iy2 - iy1)
        inter = iw * ih
        uni = (a["x2"] - a["x1"]) * (a["y2"] - a["y1"]) + (b["x2"] - b["x1"]) * (b["y2"] - b["y1"]) - inter
        return inter / uni if uni > 0 else 0.0

    by_img = {}
    for c in crops:
        by_img.setdefault(c["image"], []).append(c)
    tr = []
    for d in dets:
        if d["split"] != "train":
            continue
        best = None
        for c in by_img.get(d["image"], []):
            v = iou(d, c)
            if v > 0.5 and (best is None or v > best[0]):
                best = (v, c)
        if best is not None:
            r = dict(d)
            r["cls"] = best[1]["cls"]
            tr.append(r)
    crops = tr
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
