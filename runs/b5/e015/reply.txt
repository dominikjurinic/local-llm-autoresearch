    class FocalLoss(nn.Module):
        def __init__(self, gamma=2.0):
            super().__init__()
            self.gamma = gamma
        def forward(self, inputs, targets):
            ce = nn.functional.cross_entropy(inputs, targets, reduction='none')
            pt = torch.exp(-ce)
            return ((1 - pt) ** self.gamma * ce).mean()
    crit = FocalLoss(gamma=2.0)
