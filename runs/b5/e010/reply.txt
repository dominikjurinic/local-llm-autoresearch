    # Class-balanced training set: repeat each class's boxes up to (capped 3x) the most frequent class
    # so every batch sees all four defect types equally; the weak/rare classes get more exposure per step.
    from collections import Counter
    cls_counts = Counter(r["cls"] for r in crops)
    max_c = max(cls_counts.values())
    idx = []
    for c in sorted(cls_counts):
        rep = min(max_c, 3 * cls_counts[c]) // cls_counts[c]
        idx += [i for i, r in enumerate(crops) if r["cls"] == c] * rep
    print("class counts", dict(cls_counts), "-> balanced train size", len(idx), flush=True)
