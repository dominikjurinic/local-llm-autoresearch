    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
    images = {p: np.asarray(Image.open(os.path.join(args.data, "images", p)).convert("RGB")) for p in needed}

    # Training set = stage-1 detector boxes (train split) that overlap a labelled box:
    # same box distribution as the eval boxes we must classify.
    by_img = {}
    for r in crops:
        by_img.setdefault(r["image"], []).append(r)
    tr = []
    for d in dets:
        if d["split"] != "train":
            continue
        best, best_iou = None, 0.0
        for r in by_img.get(d["image"], ()):
            ix1, iy1 = max(d["x1"], r["x1"]), max(d["y1"], r["y1"])
            ix2, iy2 = min(d["x2"], r["x2"]), min(d["y2"], r["y2"])
            inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
            if inter <= 0:
                continue
            iou = inter / ((d["x2"] - d["x1"]) * (d["y2"] - d["y1"]) + (r["x2"] - r["x1"]) * (r["y2"] - r["y1"]) - inter)
            if iou > best_iou:
                best, best_iou = r, iou
        if best is not None and best_iou > 0.5:
            tr.append({"image": d["image"], "x1": d["x1"], "y1": d["y1"], "x2": d["x2"], "y2": d["y2"], "cls": best["cls"]})
    crops = tr
    print(f"train {len(crops)} detector boxes, eval {len(ev)} detections, {len(images)} images ({time.time() - T0:.0f}s)", flush=True)
