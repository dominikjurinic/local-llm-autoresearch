    # Background class: train-split detector boxes that do not overlap any true box (IoU <= 0.5),
    # subsampled to BG_RATIO per true box. These are the detector's own false alarms, so they match
    # the distribution of eval false alarms far better than random image regions would.
    by_img = {}
    for r in crops:
        by_img.setdefault(r["image"], []).append(r)
    bg = []
    for d in dets:
        if d["split"] != "train":
            continue
        hit = False
        for r in by_img.get(d["image"], ()):
            ix1, iy1 = max(d["x1"], r["x1"]), max(d["y1"], r["y1"])
            ix2, iy2 = min(d["x2"], r["x2"]), min(d["y2"], r["y2"])
            inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
            if inter > 0.5 * ((d["x2"] - d["x1"]) * (d["y2"] - d["y1"]) + (r["x2"] - r["x1"]) * (r["y2"] - r["y1"])):
