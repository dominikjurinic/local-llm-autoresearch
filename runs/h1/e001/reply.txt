    gt = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]

    # Stage-2 trains on stage-1's own proposals (the train-split detector boxes), labelled by
    # IoU>0.5 overlap with the true boxes, so the train box distribution matches the test one.
    def iou(a, b):
        ix1, iy1 = max(a["x1"], b["x1"]), max(a["y1"], b["y1"])
        ix2, iy2 = min(a["x2"], b["x2"]), min(a["y2"], b["y2"])
        iw, ih = max(0.0, ix2 - ix1), max(0.0, iy2 - iy1)
        inter = iw * ih
        ua = (a["x2"] - a["x1"]) * (a["y2"] - a["y1"]) + (b["x2"] - b["x1"]) * (b["y2"] - b["y1"]) - inter
        return inter / ua if ua > 0 else 0.0

    gt_by_img = {}
    for r in gt:
        gt_by_img.setdefault(r["image"], []).append(r)
    crops = []
    for d in dets:
        if d["split"] != "train":
            continue
        best, biou = None, 0.0
        for r in gt_by_img.get(d["image"], []):
            o = iou(d, r)
            if o > biou:
                best, biou = r, o
        if best is not None and biou > 0.5:
            crops.append({"image": d["image"], "x1": d["x1"], "y1": d["y1"],
                          "x2": d["x2"], "y2": d["y2"], "cls": best["cls"]})
