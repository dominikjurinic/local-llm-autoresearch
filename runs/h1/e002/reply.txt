    # Per-box geometry (log width/height/aspect/area, image-normalised) standardised on the
    # train boxes: a class-discriminative signal the pixel-only classifier never had.
    def box_geom(r):
        w = max(r["x2"] - r["x1"], 1e-6)
        h = max(r["y2"] - r["y1"], 1e-6)
        iw, ih = max(r.get("img_w", 1.0), 1.0), max(r.get("img_h", 1.0), 1.0)
        return [math.log(w / iw), math.log(h / ih), math.log(w / h), math.log((w * h) / (iw * ih))]

    tr_geom = np.array([box_geom(r) for r in crops], dtype=np.float32)
    G_MEAN, G_STD = tr_geom.mean(0), np.maximum(tr_geom.std(0), 1e-6)
