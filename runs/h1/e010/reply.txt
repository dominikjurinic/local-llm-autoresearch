    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]
    # Extra training rows: detector boxes on train images that overlap a GT box (IoU>0.5), labelled by that box's class.
    by_img = {}
    for r in crops:
        by_img.setdefault(r["image"], []).append(r)

    def iou(a, b):
        ix1, iy1 = max(a["x1"], b["x1"]), max(a["y1"], b["y1"])
        ix2, iy2 = min(a["x2"], b["x2"]), min(a["y2"], b["y2"])
        inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
        ua = (a["x2"] - a["x1"]) * (a["y2"] - a["y1"]) + (b["x2"] - b["x1"]) * (b["y2"] - b["y1"]) - inter
        return inter / ua if ua > 0 else 0.0

    extra = []
    for d in dets:
        if d["split"] != "train":
            continue
        best_c, best_i = -1, 0.0
        for r in by_img.get(d["image"], []):
            v = iou(d, r)
            if v > best_i:
                best_c, best_i = r["cls"], v
        if best_i > 0.5:
            extra.append({"image": d["image"], "x1": d["x1"], "y1": d["y1"], "x2": d["x2"], "y2": d["y2"], "cls": best_c})
    crops = crops + extra
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
