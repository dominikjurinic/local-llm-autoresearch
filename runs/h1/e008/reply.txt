class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, images, train, gstats):
        self.rows, self.images, self.train = rows, images, train
        self.gmean, self.gstd = gstats  # (mean, std) of (log area, log aspect) of training boxes

    def __len__(self):
        return len(self.rows)

    def _geom(self, r):
        w = max(r["x2"] - r["x1"], 1.0)
        h = max(r["y2"] - r["y1"], 1.0)
        g = torch.zeros(2, IMG_SIZE, IMG_SIZE)
        g[0] = (math.log(w * h) - self.gmean[0]) / self.gstd[0]
        g[1] = (math.log(w / h) - self.gmean[1]) / self.gstd[1]
        return g.clamp(-4.0, 4.0)
