class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, images, train, gstats):
        self.rows, self.images, self.train = rows, images, train
        self.gmean, self.gstd = gstats  # (mean, std) of (log area, log aspect) from training boxes

    def __len__(self):
        return len(self.rows)

    def _geom(self, r):
        w = max(r["x2"] - r["x1"], 1.0)
        h = max(r["y2"] - r["y1"], 1.0)
        a = math.log(w * h)
        b = math.log(w / h)
        g = torch.full((2, IMG_SIZE, IMG_SIZE), 0.0)
        g[0] = (a - self.gmean[0]) / self.gstd[0]
        g[1] = (b - self.gmean[1]) / self.gstd[1]
        return g.clamp(-4.0, 4.0)

    def __getitem__(self, i):
        r = self.rows[i]
        img = self.images[r["image"]]
        fh = fv = False
        bright = 1.0
        if self.train:  # same flips and brightness for both views
            fh, fv, bright = random.random() < 0.5, random.random() < 0.5, random.uniform(0.8, 1.2)
        views = []
        for ctx in (TIGHT_CONTEXT, CONTEXT):
            x = make_crop(img, r, ctx, self.train)
            if fh:
                x = x[:, ::-1]
            if fv:
                x = x[::-1, :]
            if self.train:
                x = np.clip(x * bright, 0, 1)
            x = (x - MEAN) / STD
            views.append(torch.cat([torch.from_numpy(np.ascontiguousarray(x.transpose(2, 0, 1))), self._geom(r)]))
        return views[0], views[1], r.get("cls", -1)
