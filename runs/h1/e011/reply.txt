    # detector-quality crop of the same defect, matching the box distribution it is scored on at test time.
    by_img = {}
    for r in crops:
        by_img.setdefault(r["image"], []).append(r)

    def iou(a, b):
        ix1, iy1 = max(a["x1"], b["x1"]), max(a["y1"], b["y1"])
        ix2, iy2 = min(a["x2"], b["x2"]), min(a["y2"], b["y2"])
        inter = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
        ua = (a["x2"] - a["x1"]) * (a["y2"] - a["y1"]) + (b["x2"] - b["x1"]) * (b["y2"] - b["y1"]) - inter
        return inter / ua if ua > 0 else 0.0

    train_rows = list(crops)
    for d in dets:
        if d["split"] != "train":
            continue
        best_r, best_i = None, 0.0
        for r in by_img.get(d["image"], []):
            v = iou(d, r)
            if v > best_i:
                best_r, best_i = r, v
        if best_r is not None:

class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, images, train):
        self.rows, self.images, self.train = rows, images, train

    def __len__(self):
        return len(self.rows)

    def _crop(self, img, box, ctx):
        return make_crop(img, box, ctx, self.train)

    def __getitem__(self, i):
        r = self.rows[i]
        img = self.images[r["image"]]
        fh = fv = False
        bright = 1.0
        if self.train:  # same flips and brightness for both views
            fh, fv, bright = random.random() < 0.5, random.random() < 0.5, random.uniform(0.8, 1.2)

        def finish(x):
            if fh:
                x = x[:, ::-1]
            if fv:
                x = x[::-1, :]
            if self.train:
                x = np.clip(x * bright, 0, 1)
            x = (x - MEAN) / STD
            return torch.from_numpy(np.ascontiguousarray(x.transpose(2, 0, 1)))

        views = [finish(self._crop(img, r, ctx)) for ctx in (TIGHT_CONTEXT, CONTEXT)]
        gt = r.get("gt")
        if self.train and gt is not None:  # extra view: the detector box of the same defect
            views += [finish(self._crop(img, gt, ctx)) for ctx in (TIGHT_CONTEXT, CONTEXT)]
        else:
            views += [views[0], views[1]]
        return views[0], views[1], views[2], views[3], r.get("cls", -1)
