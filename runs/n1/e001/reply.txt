class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, images, train):
        self.rows, self.images, self.train = rows, images, train

    def __len__(self):
        return len(self.rows)

    def __getitem__(self, i):
        r = self.rows[i]
        img = self.images[r["image"]]
        fh = fv = False
        bright = 1.0
        if self.train:  # same flips and brightness for both views
            fh, fv, bright = random.random() < 0.5, random.random() < 0.5, random.uniform(0.8, 1.2)
        views = []
        for ctx in (TIGHT_CONTEXT, CONTEXT):
            x = make_crop(img, r, ctx, self.train)
            if fh:
                x = x[:, ::-1]
            if fv:
                x = x[::-1, :]
            if self.train:
                x = np.clip(x * bright, 0, 1)
            x = (x - MEAN) / STD
            views.append(torch.from_numpy(np.ascontiguousarray(x.transpose(2, 0, 1))))
        # Box geometry features: log-area and aspect ratio (width/height)
        w = max(1.0, r["x2"] - r["x1"])
        h = max(1.0, r["y2"] - r["y1"])
        log_area = math.log(w * h)
        aspect = w / h
        geo = torch.tensor([log_area, aspect], dtype=torch.float32)
        return views[0], views[1], geo, r.get("cls", -1)

    class GeoHead(nn.Module):
        def __init__(self, in_dim, num_classes=4):
            super().__init__()
            self.fc = nn.Linear(in_dim + 2, num_classes)  # +2 for log_area, aspect
        def forward(self, feats, geo):
            x = torch.cat([feats, geo], dim=1)
            return self.fc(x)
    model = nn.Sequential(backbone, GeoHead(feat_dim))
