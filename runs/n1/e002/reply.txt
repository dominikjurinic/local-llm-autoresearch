    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]
    dt_tr = [d for d in dets if d["split"] == "train"]

    # Match detector train boxes to ground-truth crops by IoU > 0.5 and add as extra training data
    crops_by_image = {}
    for c in crops:
        crops_by_image.setdefault(c["image"], []).append(c)

    def iou(box1, box2):
        x1 = max(box1["x1"], box2["x1"])
        y1 = max(box1["y1"], box2["y1"])
        x2 = min(box1["x2"], box2["x2"])
        y2 = min(box1["y2"], box2["y2"])
        inter = max(0, x2 - x1) * max(0, y2 - y1)
        a1 = (box1["x2"] - box1["x1"]) * (box1["y2"] - box1["y1"])
        a2 = (box2["x2"] - box2["x1"]) * (box2["y2"] - box2["y1"])
        return inter / (a1 + a2 - inter) if (a1 + a2 - inter) > 0 else 0.0

    for d in dt_tr:
        gts = crops_by_image.get(d["image"], [])
        for gt in gts:
            if iou(d, gt) > 0.5:
                # Use detector box coordinates but ground-truth class
                crops.append({"x1": d["x1"], "y1": d["y1"], "x2": d["x2"], "y2": d["y2"],
                              "img_w": d["img_w"], "img_h": d["img_h"],
                              "image": d["image"], "cls": gt["cls"], "res": "det"})
                break
    print(f"added {len(crops) - len([r for r in read_csv(os.path.join(args.data, 'crops.csv')) if r['res'] in TRAIN_ON])} detector-matched boxes", flush=True)
