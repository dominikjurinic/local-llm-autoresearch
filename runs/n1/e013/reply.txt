class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, images, train, area_threshold=None):
        self.rows, self.images, self.train = rows, images, train
        self.contexts = (TIGHT_CONTEXT, CONTEXT, THIRD_CONTEXT)
        self.area_threshold = area_threshold
        if area_threshold is not None:
            self.is_small = []
            for r in rows:
                area = (r["x2"] - r["x1"]) * (r["y2"] - r["y1"])
                self.is_small.append(area < area_threshold)
        else:
            self.is_small = [False] * len(rows)

    def __len__(self):
        return len(self.rows)

    def __getitem__(self, i):
        r = self.rows[i]
        img = self.images[r["image"]]
        fh = fv = False
        bright = 1.0
        if self.train:  # same flips and brightness for all views
            fh, fv, bright = random.random() < 0.5, random.random() < 0.5, random.uniform(0.8, 1.2)
        views = []
        for ctx in self.contexts:
            x = make_crop(img, r, ctx, self.train)
            if fh:
                x = x[:, ::-1]
            if fv:
                x = x[::-1, :]
            if self.train:
                x = np.clip(x * bright, 0, 1)
            x = (x - MEAN) / STD
            views.append(torch.from_numpy(np.ascontiguousarray(x.transpose(2, 0, 1))))
        is_small = self.is_small[i]
        if self.train:
            # 2-of-3 random sampling: same compute as 2 views
            idx = random.sample(range(3), 2)
            return views[idx[0]], views[idx[1]], is_small, r.get("cls", -1)
        else:
            # inference: return all 3 views + is_small flag
            return views[0], views[1], views[2], is_small, r.get("cls", -1)
