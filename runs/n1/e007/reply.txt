class TwoCropNet(nn.Module):
    """Backbone (shared) + small head that takes averaged features + box metadata."""
    def __init__(self, backbone_name, pretrained, num_classes, feat_dim=None):
        super().__init__()
        self.backbone = timm.create_model(backbone_name, pretrained=pretrained, num_classes=0)
        if feat_dim is None:
            feat_dim = self.backbone.num_features
        self.head = nn.Linear(feat_dim + 2, num_classes)  # +2 for log_rel_area, aspect_ratio

    def forward(self, x1, x2, box_feats):
        # x1, x2: (B, C, H, W) two views
        f1 = self.backbone(x1)
        f2 = self.backbone(x2)
        f = (f1 + f2) * 0.5
        f = torch.cat([f, box_feats], dim=1)
        return self.head(f)

def compute_box_feats(r):
    """Return tensor [log_rel_area, aspect_ratio] for a row with x1,y1,x2,y2,img_w,img_h."""
    w = max(1.0, r["x2"] - r["x1"])
    h = max(1.0, r["y2"] - r["y1"])
    img_w = max(1.0, r["img_w"])
    img_h = max(1.0, r["img_h"])
    rel_area = (w * h) / (img_w * img_h)
    aspect = w / h
    log_rel_area = math.log(rel_area + 1e-8)
    return torch.tensor([log_rel_area, aspect], dtype=torch.float32)
