class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, images, train):
        self.rows, self.images, self.train = rows, images, train
        self.contexts = (TIGHT_CONTEXT, CONTEXT, THIRD_CONTEXT)

    def __len__(self):
        return len(self.rows)

    def __getitem__(self, i):
        r = self.rows[i]
        img = self.images[r["image"]]
        fh = fv = False
        bright = 1.0
        if self.train:  # same flips and brightness for all views
            fh, fv, bright = random.random() < 0.5, random.random() < 0.5, random.uniform(0.8, 1.2)
        views = []
        for ctx in self.contexts:
            x = make_crop(img, r, ctx, self.train)
            if fh:
                x = x[:, ::-1]
            if fv:
                x = x[::-1, :]
            if self.train:
                x = np.clip(x * bright, 0, 1)
            x = (x - MEAN) / STD
            views.append(torch.from_numpy(np.ascontiguousarray(x.transpose(2, 0, 1))))
        if self.train:
            # 2-of-3 random sampling: same compute as 2 views
            idx = random.sample(range(3), 2)
            return views[idx[0]], views[idx[1]], r.get("cls", -1)
        else:
            # inference: return all 3 views
            return views[0], views[1], views[2], r.get("cls", -1)

    # --- Add detector boxes from train split matched to GT (IoU>0.5) ---
    def compute_iou(box1, box2):
        x1 = max(box1[0], box2[0])
        y1 = max(box1[1], box2[1])
        x2 = min(box1[2], box2[2])
        y2 = min(box1[3], box2[3])
        inter = max(0, x2 - x1) * max(0, y2 - y1)
        area1 = (box1[2] - box1[0]) * (box1[3] - box1[1])
        area2 = (box2[2] - box2[0]) * (box2[3] - box2[1])
        union = area1 + area2 - inter
        return inter / union if union > 0 else 0.0
