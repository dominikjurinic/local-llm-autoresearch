```python
class GeoModel(nn.Module):
    def __init__(self, backbone, num_classes, geo_dim=2):
        super().__init__()
        self.backbone = backbone
        self.geo_fc = nn.Linear(geo_dim, 64)
        self.head = nn.Linear(backbone.num_features + 64, num_classes)
    
    def forward(self, x, geo):
        feat = self.backbone.forward_features(x)
        feat = self.backbone.global_pool(feat).flatten(1)
        geo_feat = F.relu(self.geo_fc(geo))
        combined = torch.cat([feat, geo_feat], dim=1)
        return self.head(combined)
```
