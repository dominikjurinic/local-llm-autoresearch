def box_geometry(r):
    """Return (log_area, log_aspect) for a box dict with x1,y1,x2,y2."""
    w = max(1.0, r["x2"] - r["x1"])
    h = max(1.0, r["y2"] - r["y1"])
    area = w * h
    aspect = w / h
    return np.array([math.log(area), math.log(aspect)], dtype=np.float32)

    class GeoHead(nn.Module):
        def __init__(self, in_dim, geo_dim, hidden, num_classes):
            super().__init__()
            self.geo_fc = nn.Linear(geo_dim, hidden)
            self.head = nn.Linear(in_dim + hidden, num_classes)

        def forward(self, feats, geo):
            g = F.relu(self.geo_fc(geo))
            combined = torch.cat([feats, g], dim=1)
