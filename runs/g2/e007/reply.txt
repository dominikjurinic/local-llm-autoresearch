    feat_dim = timm.get_model_attr(MODEL, "num_features")
    class TwoViewNet(nn.Module):
        def __init__(self, backbone, feat_dim, nclass=4, hidden=128):
            super().__init__()
            self.backbone = backbone
            self.mlp_t = nn.Sequential(nn.Linear(feat_dim, hidden), nn.ReLU(), nn.Linear(hidden, hidden))
            self.mlp_w = nn.Sequential(nn.Linear(feat_dim, hidden), nn.ReLU(), nn.Linear(hidden, hidden))
            self.head = nn.Linear(hidden * 3, nclass)
        def features(self, x):
            return self.backbone(x)
        def forward(self, xt, xw):
            ft, fw = self.features(xt), self.features(xw)
            ft, fw = self.mlp_t(ft), self.mlp_w(fw)
            return self.head(torch.cat([ft + fw, ft - fw, ft * fw], dim=1))
