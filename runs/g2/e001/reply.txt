    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]

    # Stage-1 boxes on the train split: label each by the true box it overlaps (IoU > 0.5) and add to training.
    tr_dets = [d for d in dets if d["split"] == "train"]
    crop_by_img, det_by_img = {}, {}
    for i, r in enumerate(crops):
        crop_by_img.setdefault(r["image"], []).append(i)
    for i, d in enumerate(tr_dets):
        det_by_img.setdefault(d["image"], []).append(i)
    added = 0
    for img, ci in crop_by_img.items():
        di = det_by_img.get(img)
        if not di:
            continue
        c = np.array([[crops[j]["x1"], crops[j]["y1"], crops[j]["x2"], crops[j]["y2"]] for j in ci])
        dt = np.array([[tr_dets[j]["x1"], tr_dets[j]["y1"], tr_dets[j]["x2"], tr_dets[j]["y2"]] for j in di])
        ix1 = np.maximum(dt[:, None, 0], c[None, :, 0]); iy1 = np.maximum(dt[:, None, 1], c[None, :, 1])
        ix2 = np.minimum(dt[:, None, 2], c[None, :, 2]); iy2 = np.minimum(dt[:, None, 3], c[None, :, 3])
        inter = np.clip(ix2 - ix1, 0.0) * np.clip(iy2 - iy1, 0.0)
        union = ((dt[:, 2] - dt[:, 0]) * (dt[:, 3] - dt[:, 1]))[:, None] + \
                ((c[:, 2] - c[:, 0]) * (c[:, 3] - c[:, 1]))[None, :] - inter
        iou = inter / np.maximum(union, 1e-9)
        k = np.argmax(iou, axis=1)
        for p in np.where(iou[np.arange(len(di)), k] > 0.5)[0]:
            d = tr_dets[di[p]]
            crops.append({"image": d["image"], "x1": d["x1"], "y1": d["y1"], "x2": d["x2"], "y2": d["y2"],
                          "cls": crops[ci[k[p]]]["cls"]})
            added += 1
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
    images = {p: np.asarray(Image.open(os.path.join(args.data, "images", p)).convert("RGB")) for p in needed}
    print(f"train {len(crops)} boxes ({added} from stage-1 dets), eval {len(ev)} detections, "
          f"{len(images)} images ({time.time() - T0:.0f}s)", flush=True)
