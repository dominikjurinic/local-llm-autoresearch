    # Polyak/SWA-style weight averaging: after warm-up, keep an EMA of the weights and
    # use the EMA for predictions (AdamW keeps training the raw weights).
    EMA_DECAY = 0.999
    ema = None
    for p in model.parameters():
        p.detach_()
    def update_ema():
        nonlocal ema
        if ema is None:
            ema = {k: v.clone() for k, v in model.state_dict().items()}
        else:
            with torch.no_grad():
                for k, v in model.state_dict().items():
                    if v.dtype.is_floating_point:
                        ema[k].mul_(EMA_DECAY).add_(v, alpha=1 - EMA_DECAY)
                    else:
                        ema[k].copy_(v)
    def load_ema():
        if ema is not None:
            model.load_state_dict(ema)
