def det_train_rows(crops, dets):
    # Detector boxes on training images, labelled by the true box they overlap (IoU > 0.5).
    det_rows = [d for d in dets if d["split"] == "train"]
    if not det_rows:
        return []
    out = []
    for img in {d["image"] for d in det_rows}:
        cs = [c for c in crops if c["image"] == img]
        ds = [d for d in det_rows if d["image"] == img]
        if not cs or not ds:
            continue
        cx1 = np.array([c["x1"] for c in cs]); cy1 = np.array([c["y1"] for c in cs])
        cx2 = np.array([c["x2"] for c in cs]); cy2 = np.array([c["y2"] for c in cs])
        dx1 = np.array([d["x1"] for d in ds]); dy1 = np.array([d["y1"] for d in ds])
        dx2 = np.array([d["x2"] for d in ds]); dy2 = np.array([d["y2"] for d in ds])
        ix1 = np.maximum(dx1[:, None], cx1[None, :]); iy1 = np.maximum(dy1[:, None], cy1[None, :])
        ix2 = np.minimum(dx2[:, None], cx2[None, :]); iy2 = np.minimum(dy2[:, None], cy2[None, :])
        inter = np.clip(ix2 - ix1, 0, None) * np.clip(iy2 - iy1, 0, None)
        area_d = (dx2 - dx1) * (dy2 - dy1)
        area_c = (cx2 - cx1) * (cy2 - cy1)
        iou = inter / np.maximum(area_d[:, None] + area_c[None, :] - inter, 1e-9)
        best = iou.argmax(1)
        for i, j in enumerate(best):
            if iou[i, j] > 0.5:
                r = dict(ds[i])
                r["cls"] = cs[j]["cls"]
                r["w"] = 0.5
                out.append(r)
    return out
