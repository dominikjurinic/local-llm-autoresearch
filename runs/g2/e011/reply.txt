class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, images, tight, wide, labels, train):
        # For train rows, tight/wide are precomputed 160px arrays (crop work is done).
        # For eval rows, tight/wide are None and the crops are computed once and cached here.
        self.rows, self.images, self.tight, self.wide, self.labels, self.train = rows, images, tight, wide, labels, train
        self._ev_cache = [None] * len(rows)

    def __len__(self):
        return len(self.labels)

    def _views(self, i):
        if self.tight is not None:
            t, w = self.tight[i], self.wide[i]
        else:
            cached = self._ev_cache[i]
            if cached is None:
                r = self.rows[i]
                img = self.images[r["image"]]
                cached = (make_crop(img, r, TIGHT_CONTEXT), make_crop(img, r, CONTEXT))
                self._ev_cache[i] = cached
            t, w = cached
        return t, w

    def __getitem__(self, i):
        fh = fv = False
        bright = 1.0
        if self.train:  # same flips and brightness for both views
            fh, fv, bright = random.random() < 0.5, random.random() < 0.5, random.uniform(0.8, 1.2)
        views = []
        for x in self._views(i):
            if fh:
                x = x[:, ::-1]
            if fv:
                x = x[::-1, :]
            if self.train:
                x = np.clip(x * bright, 0, 1)
            x = (x - MEAN) / STD
            views.append(torch.from_numpy(np.ascontiguousarray(x.transpose(2, 0, 1))))
        return views[0], views[1], self.labels[i]
