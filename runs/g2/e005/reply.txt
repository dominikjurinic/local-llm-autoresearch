# Measured stage-1 box error: for each train det box that overlaps a true box (IoU>0.5),
# record the center offset (in units of the true box long side) and the side scale ratio.
# These are used to jitter true training boxes to match the detector's real placement error.
DET_SHIFT = [0.0, 0.0]      # (dx, dy) sampled per box
DET_SCALE = 1.0
def _fit_det_error(crops, dets, images):
    tr = [d for d in dets if d["split"] == "train"]
    shifts, scales = [], []
    for img in {r["image"] for r in crops} & {d["image"] for d in tr}:
        cb = [r for r in crops if r["image"] == img]
        db = [d for d in tr if d["image"] == img]
        if not cb or not db:
            continue
        cx = np.array([(r["x1"] + r["x2"]) / 2 for r in cb])
        cy = np.array([(r["y1"] + r["y2"]) / 2 for r in cb])
        cl = np.array([max(r["x2"] - r["x1"], r["y2"] - r["y1"]) for r in cb])
        dx = np.array([(d["x1"] + d["x2"]) / 2 for d in db])
        dy = np.array([(d["y1"] + d["y2"]) / 2 for d in db])
        dl = np.array([max(d["x2"] - d["x1"], d["y2"] - d["y1"]) for d in db])
        # IoU between each det box and each true box (vectorised)
        ox1 = np.minimum(cx[:, None], dx[None, :]); oy1 = np.minimum(cy[:, None], dy[None, :])
        ox2 = np.maximum(cx[:, None], dx[None, :]); oy2 = np.maximum(cy[:, None], dy[None, :])
        iw = np.maximum(ox2 - ox1, 0); ih = np.maximum(oy2 - oy1, 0)
        inter = iw * ih
        uni = cl[:, None] * dl[None, :] + dl[None, :] * cl[:, None] - inter
        iou = np.where(uni > 0, inter / np.maximum(uni, 1e-9), 0)
        m = iou > 0.5
        if not m.any():
            continue
        ii, jj = np.where(m)
        for i, j in zip(ii, jj):
            shifts.append(((dx[j] - cx[i]) / max(cl[i], 1e-6), (dy[j] - cy[i]) / max(cl[i], 1e-6)))
            scales.append(dl[j] / max(cl[i], 1e-6))
    global DET_SHIFT, DET_SCALE
    if shifts:
        sx = np.clip(np.percentile([s[0] for s in shifts], [10, 90]), -1, 1)
        sy = np.clip(np.percentile([s[1] for s in shifts], [10, 90]), -1, 1)
        DET_SHIFT = [float(sx[1] - sx[0]), float(sy[1] - sy[0])]  # half-ranges
        DET_SCALE = float(np.clip(np.percentile(scales, 90), 0.6, 2.0))
    print(f"det-error fit: shift half-range {DET_SHIFT}, scale90 {DET_SCALE:.2f}", flush=True)
    return shifts, scales
