```python
def random_erasing(x, prob=0.5, scale=(0.05, 0.15)):
    """x is HxWx3 float32 in [0,1]"""
    if random.random() > prob:
        return x
    h, w = x.shape[:2]
    area = h * w
    for _ in range(10):  # try up to 10 times to find a valid rectangle
        eraser_area = random.uniform(*scale) * area
        eh = random.randint(1, int(math.sqrt(eraser_area)))
        ew = random.randint(1, int(math.sqrt(eraser_area)))
        if eh * ew < eraser_area * 0.5 or eh * ew > eraser_area * 2:
            continue
        ex = random.randint(0, w - ew)
        ey = random.randint(0, h - eh)
        x[ey:ey+eh, ex:ex+ew] = random.random()
        return x
    return x
```
