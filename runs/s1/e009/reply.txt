    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))

    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
    images = {p: np.asarray(Image.open(os.path.join(args.data, "images", p)).convert("RGB")) for p in needed}

    # Train on the stage-1 detector boxes themselves (what we classify at test time), labelled by
    # overlap with ground truth. This matches the test-time input distribution exactly.
    def iou(a, b):
        ix1, iy1 = max(a["x1"], b["x1"]), max(a["y1"], b["y1"])
        ix2, iy2 = min(a["x2"], b["x2"]), min(a["y2"], b["y2"])
        iw, ih = max(0.0, ix2 - ix1), max(0.0, iy2 - iy1)
        inter = iw * ih
        ua = (a["x2"] - a["x1"]) * (a["y2"] - a["y1"]) + (b["x2"] - b["x1"]) * (b["y2"] - b["y1"]) - inter
        return inter / ua if ua > 0 else 0.0

    gt_by_img = {}
    for r in crops:
        gt_by_img.setdefault(r["image"], []).append(r)
    det_tr = []
    for d in dets:
        if d["split"] != "train":
            continue
        best, best_iou = None, 0.0
        for g in gt_by_img.get(d["image"], []):
            i = iou(d, g)
            if i > best_iou:
                best, best_iou = g, i
        if best_iou > 0.5:
            row = dict(d)
            row["cls"] = best["cls"]
            det_tr.append(row)
    crops = det_tr
    print(f"train {len(crops)} detector boxes, eval {len(ev)} detections, {len(images)} images ({time.time() - T0:.0f}s)", flush=True)
