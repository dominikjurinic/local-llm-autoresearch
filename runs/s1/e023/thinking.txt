class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, images, train):
        self.rows, self.images, self.train = rows, images, train

    def __len__(self):
        return len(self.rows)

    def __getitem__(self, i):
        r = self.rows[i]
        img = self.images[r["image"]]
        fh = fv = False
        bright = 1.0
        if self.train:  # same flips and brightness for both views
            fh, fv, bright = random.random() < 0.5, random.random() < 0.5, random.uniform(0.8, 1.2)
        views = []
        for ctx in (TIGHT_CONTEXT, CONTEXT):
            x = make_crop(img, r, ctx, train)
            if fh:
                x = x[:, ::-1]
            if fv:
                x = x[::-1, :]
            if self.train:
                x = np.clip(x * bright, 0, 1)
            x = (x - MEAN) / STD
            views.append(torch.from_numpy(np.ascontiguousarray(x.transpose(2, 0, 1))))
        return views[0], views[1], r.get("cls", -1)


class TwoCropHead(nn.Module):
    """Learned per-class combination of the tight and wide crop features."""
    def __init__(self, feat_dim, hidden=256):
        super().__init__()
        self.feat_dim = feat_dim
        self.head = nn.Sequential(
            nn.Linear(feat_dim * 2, hidden),
            nn.GELU(),
            nn.Dropout(0.1),
            nn.Linear(hidden, 4),
        )

    def forward(self, f_t, f_w):
        # features: (B, feat_dim) each; concat + learned combination -> logits
        return self.head(torch.cat([f_t, f_w], dim=-1))
