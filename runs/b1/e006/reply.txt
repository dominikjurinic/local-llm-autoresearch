    MIXUP_ALPHA = 0.4       # Dirichlet concentration for the mixup weight

    def mixup_batch(xt, xw, y):
        b = xt.size(0)
        lam = torch.empty(1).uniform_(0.0, 1.0).item()
        lam = max(lam, 1.0 - lam)
        perm = torch.randperm(b)
        xt = lam * xt + (1.0 - lam) * xt[perm]
        xw = lam * xw + (1.0 - lam) * xw[perm]
        return xt, xw, y, y[perm], lam
