    # Classify every eval detection with TTA (original + horizontal flip); one prediction per class
    # with score = detector conf x mean class probability over the two views.

    with torch.no_grad(), torch.autocast(**amp):
        for xt, xw, _ in dl_ev:
            xt = xt.to(device).to(memory_format=torch.channels_last)
            xw = xw.to(device).to(memory_format=torch.channels_last)
            # original views
            lo = (model(xt) + model(xw)) / 2
            # horizontally flipped views
            xt_f = xt.flip(3)
            xw_f = xw.flip(3)
            lo_f = (model(xt_f) + model(xw_f)) / 2
