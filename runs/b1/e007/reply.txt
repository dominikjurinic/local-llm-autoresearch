    with torch.no_grad(), torch.autocast(**amp):
        for xt, xw, _ in dl_ev:
            xt = xt.to(device).to(memory_format=torch.channels_last)
            xw = xw.to(device).to(memory_format=torch.channels_last)
            probs.append(torch.softmax(((model(xt) + model(xw)) / 2).float(), 1).cpu())
    probs = torch.cat(probs).numpy()

    # Score rows by an empirical hit-rate lookup fitted on the train split: for class c and a box with
    # (conf, p_c), the score is the fraction of train boxes in the same (conf decile, p decile) cell that
    # are true class-c boxes. This re-ranks each class's rows by estimated hit probability instead of raw
    # conf x softmax, which downweights overconfident false alarms and wrong-class rows.
    train_dets = [d for d in dets if d["split"] == "train"]
    tr_probs = np.zeros((len(train_dets), 4), dtype=np.float32)
    dl_tr_sc = torch.utils.data.DataLoader(
        BoxDataset([d for d in train_dets], images, False), batch_size=256, shuffle=False,
        num_workers=workers, pin_memory=True)
    with torch.no_grad(), torch.autocast(**amp):
        for xt, xw, _ in dl_tr_sc:
            xt = xt.to(device).to(memory_format=torch.channels_last)
            xw = xw.to(device).to(memory_format=torch.channels_last)
            tr_probs = np.vstack([tr_probs, torch.softmax(((model(xt) + model(xw)) / 2).float(), 1).cpu().numpy()]) if len(tr_probs) else torch.softmax(((model(xt) + model(xw)) / 2).float(), 1).cpu().numpy()
    tr_probs = tr_probs[:len(train_dets)]
    confs = np.array([d["conf"] for d in train_dets], dtype=np.float32)
    is_true = np.array([("cls" in d) for d in train_dets])
    tr_cls = np.array([d.get("cls", -1) for d in train_dets])
    cb = np.clip(np.searchsorted(np.quantile(confs, np.arange(1, 10) / 10), confs, side="right"), 0, 9)
    lookup = np.zeros((4, 10, 10), dtype=np.float64)
    lookup_n = np.zeros((4, 10, 10), dtype=np.float64)
    conf_only = np.zeros((4, 10), dtype=np.float64)
    conf_only_n = np.zeros((4, 10), dtype=np.float64)
    for c in range(4):
        pb = np.clip(np.searchsorted(np.quantile(tr_probs[:, c], np.arange(1, 10) / 10), tr_probs[:, c], side="right"), 0, 9)
        hit = (is_true & (tr_cls == c)).astype(np.float64)
        for i in range(len(train_dets)):
            lookup[c, cb[i], pb[i]] += hit[i]; lookup_n[c, cb[i], pb[i]] += 1
            conf_only[c, cb[i]] += hit[i]; conf_only_n[c, cb[i]] += 1
    for c in range(4):
        conf_only[c] /= np.maximum(conf_only_n[c], 1)
        for i in range(10):
            for j in range(10):
                if lookup_n[c, i, j] > 0:
                    lookup[c, i, j] = 0.8 * (lookup[c, i, j] / lookup_n[c, i, j]) + 0.2 * conf_only[c, i]
                else:
                    lookup[c, i, j] = conf_only[c, i]
