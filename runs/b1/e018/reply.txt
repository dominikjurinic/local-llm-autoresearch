    model = timm.create_model(MODEL, pretrained=PRETRAINED, num_classes=4)
    # Expand the stem from 3 to 4 channels, keeping the pretrained 3-channel weights and
    # initializing the 4th (box-size) channel as their mean, so pretrained features start intact.
    stem = model.stem
    new_stem = nn.Conv2d(
        N_IN_CH, stem.out_channels, stem.kernel_size, stride=stem.stride,
        padding=stem.padding, groups=stem.groups, bias=stem.bias is not None)
    with torch.no_grad():
        new_stem.weight[:, :3, :, :] = stem.weight
        new_stem.weight[:, 3, :, :] = stem.weight.mean(dim=1, keepdim=True)
        if stem.bias is not None:
            new_stem.bias.copy_(stem.bias)
    model.stem = new_stem
    model = model.to(device).to(memory_format=torch.channels_last)
