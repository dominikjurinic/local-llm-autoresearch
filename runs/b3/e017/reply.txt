    class SmoothFocalLoss(nn.Module):
        # Focal loss (focus on hard, confusable examples) + label smoothing (cap over-confidence).
        def __init__(self, gamma=2.0, smooth=0.1):
            super().__init__()
            self.gamma, self.smooth = gamma, smooth

        def forward(self, x, y):
            g = self.gamma
            ce = nn.functional.cross_entropy(x, y, reduction="none")
            pt = torch.exp(-ce)
            return ((1 - pt) ** g * ce).mean() + self.smooth * x.mean()

    crit = SmoothFocalLoss(gamma=2.0, smooth=0.1)
