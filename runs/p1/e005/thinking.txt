```python
class CropFuser(nn.Module):
    def __init__(self, in_dim=1280, d=512):
        super().__init__()
        self.proj_t = nn.Linear(in_dim, d)
        self.proj_w = nn.Linear(in_dim, d)
        self.act = nn.GELU()
        self.scale = math.sqrt(d)
        self.out_mlp = nn.Sequential(
            nn.Linear(d * 3, d),
            nn.GELU(),
        )
    
    def forward(self, f_t, f_w):
        q = self.act(self.proj_t(f_t))  # (B, D)
        w = self.act(self.proj_w(f_w))  # (B, D)
        # Cross-attention: element-wise relevance gating
        # Each dim of q attends to corresponding dim of w
        relevance = torch.sigmoid(q * w / self.scale)  # (B, D)
        attn = relevance * w  # (B, D) - wide features weighted by tight's need
        out = self.out_mlp(torch.cat([q, attn, w], dim=-1))  # (B, D)
        return out
```

class CropFuser(nn.Module):
    def __init__(self, in_dim=1280, d=512):
        super().__init__()
        self.proj_t = nn.Linear(in_dim, d)
        self.proj_w = nn.Linear(in_dim, d)
        self.act = nn.GELU()
        self.scale = math.sqrt(d)
        self.out_mlp = nn.Sequential(
            nn.Linear(d * 3, d),
            nn.GELU(),
        )
    
    def forward(self, f_t, f_w):
        q = self.act(self.proj_t(f_t)).unsqueeze(-1)  # (B, D, 1)
        k = self.act(self.proj_w(f_w)).unsqueeze(-1)  # (B, D, 1)
        v = k  # (B, D, 1)
        # Cross-attention: (B, D, 1) @ (B, 1, D) -> (B, D, D)
        attn = torch.softmax(q @ k.transpose(-2, -1) / self.scale, dim=-1)  # (B, D, D)
        ctx = attn @ v  # (B, D, 1)
        ctx = ctx.squeeze(-1)  # (B, D)
        q_flat = q.squeeze(-1)  # (B, D)
        k_flat = k.squeeze(-1)  # (B, D)
        out = self.out_mlp(torch.cat([q_flat, ctx, k_flat], dim=-1))  # (B, D)
        return out
