class CropFuser(nn.Module):
    """Cross-attention fusion of tight and wide crop features."""
    def __init__(self, in_dim=1280, d=512):
        super().__init__()
        self.proj_t = nn.Linear(in_dim, d)
        self.proj_w = nn.Linear(in_dim, d)
        self.act = nn.GELU()
        self.scale = math.sqrt(d)
        self.out_mlp = nn.Sequential(
            nn.Linear(d * 3, d),
            nn.GELU(),
        )

    def forward(self, f_t, f_w):
        # f_t, f_w: (B, in_dim) -> project to (B, d)
        q = self.act(self.proj_t(f_t)).unsqueeze(-1)  # (B, D, 1)
        k = self.act(self.proj_w(f_w)).unsqueeze(-1)  # (B, D, 1)
        v = k  # (B, D, 1)
        # Cross-attention: D queries attend over D keys
        attn = torch.softmax(q @ k.transpose(-2, -1) / self.scale, dim=-1)  # (B, D, D)
        ctx = (attn @ v).squeeze(-1)  # (B, D)
        q_flat = q.squeeze(-1)  # (B, D)
        k_flat = k.squeeze(-1)  # (B, D)
        return self.out_mlp(torch.cat([q_flat, ctx, k_flat], dim=-1))  # (B, D)


class TwoCropClassifier(nn.Module):
    def __init__(self, backbone, fuser, head):
        super().__init__()
        self.backbone = backbone
        self.fuser = fuser
        self.head = head

    def forward(self, x_t, x_w):
        f_t = self.backbone(x_t)  # (B, 1280)
        f_w = self.backbone(x_w)  # (B, 1280)
        fused = self.fuser(f_t, f_w)  # (B, 512)
        return self.head(fused)  # (B, 4)
