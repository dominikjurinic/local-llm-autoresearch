```python
class TwoCropClassifier(nn.Module):
    def __init__(self, backbone, feat_dim=1280, hidden=512, num_classes=4):
        super().__init__()
        self.backbone = backbone
        self.head = nn.Sequential(
            nn.Linear(feat_dim * 2, hidden),
            nn.GELU(),
            nn.Linear(hidden, num_classes),
        )

    def forward(self, x_t, x_w):
        f = torch.cat([self.backbone(x_t), self.backbone(x_w)], dim=-1)
        return self.head(f)
```
