Cross-attention lets the tight crop's texture features attend to wide-crop context (and vice versa) instead of being collapsed by an MLP into one vector, which should specifically fix the dominant deposit↔stain confusion. Expected gain: +0.01+ mAP on true-defect classification if it recovers ~30% of the 55+46 error cases; differs from e005/e006 (which crashed) and e007's MLP fusion by an order of magnitude in how features interact rather than another knob-turn. Note: the plan's "pooled [B,C] vectors" make a standard attention degenerate (a single token gives a weight that is always 1, so the query has no effect); the smallest fix that keeps the idea is to run the cross-attention on the backbone's spatial feature map ([B,C,5,5] → [B,25,C] tokens for 160px input) and mean-pool afterwards, so the query genuinely selects context.

class CrossAttention(nn.Module):
    """Single multi-head cross-attention layer: queries q attend to keys/values kv (both [B, L, C])."""

    def __init__(self, dim, n_heads=8):
        super().__init__()
        assert dim % n_heads == 0
        self.n_heads = n_heads
        self.head_dim = dim // n_heads
        self.w_q = nn.Linear(dim, dim)
        self.w_k = nn.Linear(dim, dim)
        self.w_v = nn.Linear(dim, dim)
        self.w_o = nn.Linear(dim, dim)

    def forward(self, q, kv):
        B, L, C = q.shape
        h, d = self.n_heads, self.head_dim
        q = self.w_q(q).view(B, L, h, d).transpose(1, 2)   # [B, h, L, d]
        k = self.w_k(kv).view(B, L, h, d).transpose(1, 2)  # [B, h, L, d]
        v = self.w_v(kv).view(B, L, h, d).transpose(1, 2)  # [B, h, L, d]
        a = torch.matmul(q, k.transpose(-2, -1)) / math.sqrt(d)  # [B, h, L, L]
        a = torch.softmax(a, dim=-1)
        out = torch.matmul(a, v)                              # [B, h, L, d]
        out = out.transpose(1, 2).reshape(B, L, C)           # [B, L, C]
        return self.w_o(out)


class TwoCropClassifier(nn.Module):
    """Shared EfficientNet-B0 backbone over tight + wide crops, bidirectional cross-attention, 4-class head."""

    def __init__(self, backbone, hidden=512, num_classes=4):
        super().__init__()
        self.backbone = backbone
        C = backbone.num_features
        self.xattn = CrossAttention(C)
        self.head = nn.Sequential(
            nn.Linear(C * 2, hidden),
            nn.GELU(),
            nn.Linear(hidden, num_classes),
        )

    def _feats(self, x):
        f = self.backbone.forward_features(x)          # [B, C, H, W]
        B, C, H, W = f.shape
        return f.permute(0, 2, 3, 1).reshape(B, H * W, C)  # [B, L, C]

    def forward(self, x_t, x_w):
        f_t = self._feats(x_t)
        f_w = self._feats(x_w)
        out_t = f_t + self.xattn(f_t, f_w)             # tight queries attend to wide context
        out_w = f_w + self.xattn(f_w, f_t)             # wide queries attend to tight texture
        f = torch.cat([out_t.mean(dim=1), out_w.mean(dim=1)], dim=-1)  # [B, 2C]
        return self.head(f)
