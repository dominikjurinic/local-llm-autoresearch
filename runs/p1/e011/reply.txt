def make_crop(img, r, context, train):
    h, w = img.shape[:2]
    cx, cy = (r["x1"] + r["x2"]) / 2, (r["y1"] + r["y2"]) / 2
    side = max(max(r["x2"] - r["x1"], r["y2"] - r["y1"]) * context, MIN_CROP)
    if train:  # light geometric jitter
        cx += random.uniform(-0.1, 0.1) * side
        cy += random.uniform(-0.1, 0.1) * side
        side *= random.uniform(0.85, 1.2)
    s = max(2, int(round(side)))
    x0, y0 = int(round(cx - s / 2)), int(round(cy - s / 2))
    canvas = np.full((s, s, 3), 114, dtype=np.uint8)  # gray padding outside the image
    sx0, sy0, sx1, sy1 = max(0, x0), max(0, y0), min(w, x0 + s), min(h, y0 + s)
    if sx1 > sx0 and sy1 > sy0:
        canvas[sy0 - y0:sy1 - y0, sx0 - x0:sx1 - x0] = img[sy0:sy1, sx0:sx1]
    return np.asarray(Image.fromarray(canvas).resize((IMG_SIZE, IMG_SIZE), Image.BILINEAR), dtype=np.float32) / 255.0

class MultiScaleClassifier(nn.Module):
    """Shared backbone over 4 scale crops, per-scale 1x1 projection, concat, MLP head."""

    def __init__(self, backbone, num_scales=4, proj_dim=256, hidden=512, num_classes=4):
        super().__init__()
        self.backbone = backbone
        C = backbone.num_features
        self.scale_projs = nn.ModuleList([nn.Conv2d(C, proj_dim, 1) for _ in range(num_scales)])
        self.head = nn.Sequential(
            nn.Linear(num_scales * proj_dim, hidden),
            nn.GELU(),
            nn.Linear(hidden, num_classes),
        )

    def forward(self, views):
        feats = []
        for i, x in enumerate(views):
            f = self.backbone.forward_features(x)   # [B, C, H, W]
            f = self.scale_projs[i](f)              # [B, proj_dim, H, W]
            f = f.mean(dim=[2, 3])                  # [B, proj_dim]
            feats.append(f)
        f = torch.cat(feats, dim=1)                 # [B, num_scales * proj_dim]
        return self.head(f)

class MultiScaleClassifier(nn.Module):
    """Shared backbone over 4 scale crops + 1 high-res crop, per-scale 1x1 projection, concat, MLP head."""

    def __init__(self, backbone, num_scales=4, proj_dim=256, hidden=512, num_classes=4):
        super().__init__()
        self.backbone = backbone
        C = backbone.num_features
        self.scale_projs = nn.ModuleList([nn.Conv2d(C, proj_dim, 1) for _ in range(num_scales)])
        self.hr_proj = nn.Conv2d(C, proj_dim, 1)
        self.head = nn.Sequential(
            nn.Linear((num_scales + 1) * proj_dim, hidden),
            nn.GELU(),
            nn.Linear(hidden, num_classes),
        )

    def forward(self, views):
        feats = []
        for i, x in enumerate(views):
            f = self.backbone.forward_features(x)   # [B, C, H, W]
            if i < len(self.scale_projs):
                f = self.scale_projs[i](f)
            else:
                f = self.hr_proj(f)
            f = f.mean(dim=[2, 3])                  # [B, proj_dim]
            feats.append(f)
        f = torch.cat(feats, dim=1)                 # [B, (num_scales+1) * proj_dim]
        return self.head(f)

class MultiScaleClassifier(nn.Module):
    """Shared backbone over 4 scale crops + 1 high-res crop, per-scale 1x1 projection,
    spatial pyramid fusion for the scales, separate high-res branch, concat, MLP head."""

    def __init__(self, backbone, num_scales=4, proj_dim=256, hidden=512, num_classes=4):
        super().__init__()
        self.backbone = backbone
        C = backbone.num_features
        self.scale_projs = nn.ModuleList([nn.Conv2d(C, proj_dim, 1) for _ in range(num_scales)])
        self.stem = nn.Sequential(
            nn.Conv2d(num_scales * proj_dim, hidden, 3, padding=1),
            nn.GELU(),
            nn.Conv2d(hidden, proj_dim, 3, padding=1),
            nn.GELU(),
        )
        self.hr_proj = nn.Conv2d(C, proj_dim, 1)
        self.head = nn.Sequential(
            nn.Linear(2 * proj_dim, hidden * 2),
            nn.GELU(),
            nn.Linear(hidden * 2, num_classes),
        )

    def forward(self, views):
        feats = []
        for i in range(len(self.scale_projs)):
            f = self.backbone.forward_features(views[i])   # [B, C, H, W]
            f = self.scale_projs[i](f)                     # [B, proj_dim, H, W]
            feats.append(f)
        f = torch.cat(feats, dim=1)                        # [B, num_scales * proj_dim, H, W]
        f = self.stem(f)                                   # [B, proj_dim, H, W]
        f = f.mean(dim=[2, 3])                             # pyramid vector [B, proj_dim]
        hr = self.backbone.forward_features(views[-1])     # [B, C, H, W]
        hr = self.hr_proj(hr)                              # [B, proj_dim, H, W]
        hr = hr.mean(dim=[2, 3])                           # high-res vector [B, proj_dim]
        f = torch.cat([f, hr], dim=1)                      # [B, 2 * proj_dim]
        return self.head(f)
