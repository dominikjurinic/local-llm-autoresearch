```python
def estimate_centre(img, r):
    h, w = img.shape[:2]
    x0 = max(0, int(r["x1"])); y0 = max(0, int(r["y1"]))
    x1 = min(w, int(r["x2"])); y1 = min(h, int(r["y2"]))
    if x1 - x0 < 2 or y1 - y0 < 2:
        return (r["x1"] + r["x2"]) / 2, (r["y1"] + r["y2"]) / 2
    region = img[y0:y1, x0:x1].astype(np.float32)
    # gradient magnitude via simple finite differences
    gx = np.zeros_like(region)
    gy = np.zeros_like(region)
    gx[:, 1:-1] = (region[:, 2:] - region[:, :-2]) / 2.0
    gy[1:-1, :] = (region[2:, :] - region[:-2, :]) / 2.0
    mag = np.sqrt((gx**2).sum(-1) + (gy**2).sum(-1))  # sum over channels
    # find max
    idx = np.argmax(mag)
    iy, ix = np.unravel_index(idx, mag.shape)
    return x0 + ix, y0 + iy
```
