def read_csv(path):
    rows = list(csv.DictReader(open(path)))
    for r in rows:
        for k in ("x1", "y1", "x2", "y2", "conf"):
            if k in r:
                r[k] = float(r[k])
        if "cls" in r:
            r["cls"] = int(r["cls"])
    return rows


def build_train_rows(crops, dets):
    # Train on the stage-1 detector boxes themselves (the test-time input distribution):
    # label = class of the crop they overlap (IoU > 0.5), else background (class 4).
    tr = [d for d in dets if d["split"] == "train"]
    by_img = {}
    for c in crops:
        by_img.setdefault(c["image"], []).append(c)
    rows = []
    for d in tr:
        cls = 4
        for c in by_img.get(d["image"], ()):
            ix = max(0.0, min(d["x2"], c["x2"]) - max(d["x1"], c["x1"]))
            iy = max(0.0, min(d["y2"], c["y2"]) - max(d["y1"], c["y1"]))
            inter = ix * iy
            if inter > 0.5 * ((d["x2"] - d["x1"]) * (d["y2"] - d["y1"])):
                cls = c["cls"]
                break
        d["cls"] = cls
        rows.append(d)
    return rows
