def iou(a, b):
    ax1, ay1, ax2, ay2 = a["x1"], a["y1"], a["x2"], a["y2"]
    bx1, by1, bx2, by2 = b["x1"], b["y1"], b["x2"], b["y2"]
    ix1, iy1 = max(ax1, bx1), max(ay1, by1)
    ix2, iy2 = min(ax2, bx2), min(ay2, by2)
    iw, ih = max(0.0, ix2 - ix1), max(0.0, iy2 - iy1)
    inter = iw * ih
    if inter <= 0:
        return 0.0
    aa = (ax2 - ax1) * (ay2 - ay1)
    ba = (bx2 - bx1) * (by2 - by1)
    return inter / (aa + ba - inter)

    ev = [d for d in dets if d["split"] == "eval"]
    # Background training boxes: train-split detector boxes that overlap no true defect.
    by_img = {}
    for r in crops:
        by_img.setdefault(r["image"], []).append(r)
    bg = []
    for d in dets:
        if d["split"] != "train":

        best = max((iou(d, g) for g in by_img.get(d["image"], [])), default=0.0)
        if best < 0.5:
            bg.append(dict(d, cls=N_CLASS - 1))
    needed = {r["image"] for r in crops} | {d["image"] for d in ev} | {b["image"] for b in bg}
    images = {p: np.asarray(Image.open(os.path.join(args.data, "images", p)).convert("RGB")) for p in needed}
    print(f"train {len(crops)} defect + {len(bg)} background boxes, eval {len(ev)} detections, {len(images)} images ({time.time() - T0:.0f}s)", flush=True)
