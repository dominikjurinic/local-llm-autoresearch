def box_size_feats(rows):
    # per-box [log1p(long_side), log(aspect)] with train-set mean/std normalization
    ls = np.array([math.log1p(max(r["x2"] - r["x1"], r["y2"] - r["y1"], 1.0)) for r in rows])
    ar = np.array([math.log(max(r["x2"] - r["x1"], 1.0) / max(r["y2"] - r["y1"], 1.0)) for r in rows])
    m = np.array([ls.mean(), ar.mean()]); s = np.array([ls.std() + 1e-6, ar.std() + 1e-6])
    return (m, s)

def size_stats(rows):
    ls = np.array([math.log1p(max(r["x2"] - r["x1"], r["y2"] - r["y1"], 1.0)) for r in rows])
    ar = np.array([math.log(max(r["x2"] - r["x1"], 1.0) / max(r["y2"] - r["y1"], 1.0)) for r in rows])
    return np.array([ls.mean(), ar.mean()]), np.array([ls.std() + 1e-6, ar.std() + 1e-6])


def size_feats(rows, m, s):
    ls = np.array([math.log1p(max(r["x2"] - r["x1"], r["y2"] - r["y1"], 1.0)) for r in rows])
    ar = np.array([math.log(max(r["x2"] - r["x1"], 1.0) / max(r["y2"] - r["y1"], 1.0)) for r in rows])
    return torch.from_numpy(((np.stack([ls, ar], axis=1) - m) / s).astype(np.float32))
