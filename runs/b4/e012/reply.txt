SIZE_LO, SIZE_HI = math.log(8.0), math.log(512.0)   # normalisation range for box size channel
ASPECT_LO, ASPECT_HI = math.log(0.2), math.log(5.0) # normalisation range for aspect channel


def box_meta(r):
    bw, bh = max(r["x2"] - r["x1"], 1e-6), max(r["y2"] - r["y1"], 1e-6)
    size = min(max((math.log(max(bw, bh)) - SIZE_LO) / (SIZE_HI - SIZE_LO), 0.0), 1.0)
    aspect = min(max((math.log(bw / bh) - ASPECT_LO) / (ASPECT_HI - ASPECT_LO), 0.0), 1.0)
    return size, aspect
