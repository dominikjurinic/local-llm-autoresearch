class FusionModel(nn.Module):
    """EfficientNet-B0 backbone with adaptive per-box fusion via MLP."""
    def __init__(self, base):
        super().__init__()
        self.base = base
        self.mlp = nn.Sequential(
            nn.Linear(3, 16),
            nn.ReLU(),
            nn.Linear(16, 1)
        )

    def forward(self, xt, xw, feat):
        logits_t = self.base(xt)
        logits_w = self.base(xw)
        w = torch.sigmoid(self.mlp(feat)).squeeze(-1).unsqueeze(-1)  # (B,1)
        return w * logits_t + (1 - w) * logits_w
