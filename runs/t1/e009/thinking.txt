    all_crops = read_csv(os.path.join(args.data, "crops.csv"))
    crops = [r for r in all_crops if r["res"] in TRAIN_ON]

    ev = [d for d in dets if d["split"] == "eval"]

    def iou(a, b):
        ix1 = max(a["x1"], b["x1"]); iy1 = max(a["y1"], b["y1"])
        ix2 = min(a["x2"], b["x2"]); iy2 = min(a["y2"], b["y2"])
        iw = max(0.0, ix2 - ix1); ih = max(0.0, iy2 - iy1)
        inter = iw * ih
        if inter <= 0.0:
            return 0.0
        union = (a["x2"] - a["x1"]) * (a["y2"] - a["y1"]) + (b["x2"] - b["x1"]) * (b["y2"] - b["y1"]) - inter
        return inter / union if union > 0 else 0.0

    gt_by_image = {}
    for r in all_crops:
        gt_by_image.setdefault(r["image"], []).append(r)
    extra_train = []
    for d in dets:
        if d["split"] != "train":
            continue
        best_iou, best_cls = 0.0, -1
        for g in gt_by_image.get(d["image"], ()):
            v = iou(d, g)
            if v > best_iou:
                best_iou, best_cls = v, g["cls"]
        if best_iou > 0.5:
            d["cls"] = best_cls
            extra_train.append(d)
    train_rows = crops + extra_train
    needed = {r["image"] for r in train_rows} | {d["image"] for d in ev}
