class FusionModel(nn.Module):
    """EfficientNet-B0 backbone with a learnable per-class tight/wide logit fusion."""
    def __init__(self, base):
        super().__init__()
        self.base = base
        self.logit_w = nn.Parameter(torch.zeros(4))  # sigmoid -> 0.5 (equal weight)

    def forward(self, xt, xw):
        w = torch.sigmoid(self.logit_w)
        return w * self.base(xt) + (1 - w) * self.base(xw)

class FusionModel(nn.Module):
    """EfficientNet-B0 backbone with learnable per-class attention over three views."""
    def __init__(self, base, n_views=3):
        super().__init__()
        self.base = base
        self.attn = nn.Parameter(torch.zeros(4, n_views))  # per-class, per-view; softmax over views

    def forward(self, views):
        stacked = torch.stack([self.base(v) for v in views], dim=1)  # (B, n_views, 4)
        w = torch.softmax(self.attn, dim=1)  # (4, n_views)
        return torch.einsum("bvc, cv -> bc", stacked, w)
