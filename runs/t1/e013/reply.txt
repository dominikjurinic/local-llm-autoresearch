    all_crops = read_csv(os.path.join(args.data, "crops.csv"))
    crops = [r for r in all_crops if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]

    # Extra training rows: train-split detector boxes that overlap a true box (IoU > 0.5),
    # cropped at the detector box geometry and labelled with the true box's class.
    crops_by_img = {}
    for r in all_crops:
        crops_by_img.setdefault(r["image"], []).append(r)
    extra = []
    for d in dets:
        if d["split"] != "train":
            continue
        ad = (d["x2"] - d["x1"]) * (d["y2"] - d["y1"])
        best, best_iou = None, 0.0
        for r in crops_by_img.get(d["image"], ()):
            iw = min(d["x2"], r["x2"]) - max(d["x1"], r["x1"])
            ih = min(d["y2"], r["y2"]) - max(d["y1"], r["y1"])
            if iw <= 0 or ih <= 0:
                continue
            inter = iw * ih
            ar = (r["x2"] - r["x1"]) * (r["y2"] - r["y1"])
            iou = inter / (ad + ar - inter)
            if iou > best_iou:
                best, best_iou = r, iou
        if best is not None and best_iou > 0.5:
            row = dict(d)
            row["cls"] = best["cls"]
            extra.append(row)
    crops = crops + extra
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
    images = {p: np.asarray(Image.open(os.path.join(args.data, "images", p)).convert("RGB")) for p in needed}
    print(f"train {len(crops)} boxes ({len(extra)} detector boxes), eval {len(ev)} detections, {len(images)} images ({time.time() - T0:.0f}s)", flush=True)
