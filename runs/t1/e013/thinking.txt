```python
    # Extra training rows: detector boxes from the train split, labelled by the GT box they overlap (IoU > 0.5).
    crops_by_img = {}
    for r in read_csv(os.path.join(args.data, "crops.csv")):
        crops_by_img.setdefault(r["image"], []).append(r)
    extra = []
    for d in dets:
        if d["split"] != "train":
            continue
        best, best_iou = None, 0.0
        for r in crops_by_img.get(d["image"], ()):
            ix1, iy1 = max(d["x1"], r["x1"]), max(d["y1"], r["y1"])
            ix2, iy2 = min(d["x2"], r["x2"]), min(d["y2"], r["y2"])
            iw, ih = max(0.0, ix2 - ix1), max(0.0, iy2 - iy1)
            inter = iw * ih
            if inter <= 0:
                continue
            area_d = (d["x2"] - d["x1"]) * (d["y2"] - d["y1"])
            area_r = (r["x2"] - r["x1"]) * (r["y2"] - r["y1"])
            iou = inter / (area_d + area_r - inter)
            if iou > best_iou:
                best, best_iou = r, iou
        if best is not None and best_iou > 0.5:
            row = dict(d)
            row["cls"] = best["cls"]
            extra.append(row)
    crops = crops + extra
```
