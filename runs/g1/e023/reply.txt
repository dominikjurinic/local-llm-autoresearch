def make_pair(img, r, train):
    # one 160x320 canvas: tight view (left) + wide view (right), same center, shared jitter
    fh, fv, bright = False, False, 1.0
    if train:
        fh, fv, bright = random.random() < 0.5, random.random() < 0.5, random.uniform(0.8, 1.2)
    pair = np.concatenate([make_crop(img, r, TIGHT_CONTEXT, train), make_crop(img, r, CONTEXT, train)], axis=1)
    if fh:
        pair = pair[:, ::-1]
    if fv:
        pair = pair[::-1, :]
    if train:
        pair = np.clip(pair * bright, 0, 1)
    pair = (pair - MEAN) / STD
    return torch.from_numpy(np.ascontiguousarray(pair.transpose(2, 0, 1)))
