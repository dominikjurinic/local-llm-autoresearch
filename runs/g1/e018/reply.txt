    crit = nn.CrossEntropyLoss()
    # Confused-class margin loss: push logit_true above logit_confused by at least MARGIN.
    # Pairs from the confusion matrix: pore(0)-stain(3), stain(3)-deposit(1), deposit(1)-stain(3)
    CONFUSED_PAIRS = [(0, 3), (0, 1), (3, 0), (3, 1), (1, 3)]
    MARGIN = 1.0
    MARGIN_WEIGHT = 0.5

    def margin_loss(logits, targets):
        # logits: (B, 4), targets: (B,)
        total = torch.tensor(0.0, device=logits.device)
        for c_true, c_conf in CONFUSED_PAIRS:
            mask = (targets == c_true)
            if mask.any():
                gap = logits[mask, c_true] - logits[mask, c_conf]
                total = total + (MARGIN - gap).clamp(min=0).sum()
        return total / max(len(targets), 1)
