    # Extra training rows: train-split detector boxes that overlap a true box (IoU>0.5),
    # labelled with that true box's class. Matches the classifier to the imperfect stage-1
    # boxes it is scored on (train/test box-quality mismatch).
    def _iou(A, B):
        ax1, ay1, ax2, ay2 = A[:, 0], A[:, 1], A[:, 2], A[:, 3]
        bx1, by1, bx2, by2 = B[:, 0], B[:, 1], B[:, 2], B[:, 3]
        ix1 = np.maximum(ax1[:, None], bx1[None, :]); iy1 = np.maximum(ay1[:, None], by1[None, :])
        ix2 = np.minimum(ax2[:, None], bx2[None, :]); iy2 = np.minimum(ay2[:, None], by2[None, :])
        inter = np.clip(ix2 - ix1, 0, None) * np.clip(iy2 - iy1, 0, None)
        union = (ax2 - ax1) * (ay2 - ay1)[:, None] + (bx2 - bx1) * (by2 - by1)[None, :] - inter
        return inter / np.clip(union, 1e-9, None)
