    # Test-time augmentation: average logits over the two views and their horizontal flips (4 views).
    model.eval()
    probs = []
    with torch.no_grad(), torch.autocast(**amp):
        for xt, xw, _ in dl_ev:
            xt = xt.to(device).to(memory_format=torch.channels_last)
            xw = xw.to(device).to(memory_format=torch.channels_last)
            acc = torch.zeros(xt.shape[0], 4, device=device)
            for a in (xt, xt.flip(3)):
                for b in (xw, xw.flip(3)):
                    acc += model(a, b).float()
            probs.append(torch.softmax(acc / 4.0, 1).cpu())
    probs = torch.cat(probs).numpy()
