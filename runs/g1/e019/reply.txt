    # Train until the budget minus the reserve is used up.
    # Phase 1 (first 70%): 5% linear warm-up, then cosine decay to SWA_FRAC * LR.
    # Phase 2 (last 30%): constant LR = SWA_FRAC * LR while maintaining a running
    # average of the weights (SWA); the averaged weights are used for prediction.
    SWA_FRAC = 0.1
    SWA_START = 0.70
    t_start = time.time()
    t_end = deadline - RESERVE_S
    budget = max(1.0, t_end - t_start)
    steps, epoch, done = 0, 0, False
    swa_avg = None
    model.train()
    while not done:
        tot, nb = 0.0, 0
        for xt, xw, y in dl_tr:
            frac = min(1.0, (time.time() - t_start) / budget)
            if frac < 0.05:
                f = frac / 0.05
            elif frac < SWA_START:
                f = SWA_FRAC + (1.0 - SWA_FRAC) * 0.5 * (1 + math.cos(math.pi * (frac - 0.05) / (SWA_START - 0.05)))
            else:
                f = SWA_FRAC
            for pg in opt.param_groups:
                pg["lr"] = LR * max(f, 1e-3)
            if frac >= SWA_START:
                if swa_avg is None:
                    swa_avg = {k: v.clone() for k, v in model.state_dict().items()}

                        swa_avg[k].mul_(0.98).add_(v, alpha=0.02)
