class TwoViewModel(nn.Module):
    def __init__(self, backbone, head, mid_idx):
        super().__init__()
        self.backbone = backbone
        self.head = head
        self.mid_idx = mid_idx
        self._mid = None

    def _hook(self, module, inp, out):
        self._mid = out

    def _feats(self, x):
        self._mid = None
        h = self.backbone.blocks[self.mid_idx].register_forward_hook(self._hook)
        f = self.backbone(x)
        h.remove()
        if self._mid is not None:
            m = self._mid.float().mean(dim=[2, 3])
            return torch.cat([f, m], dim=1)
        return torch.cat([f, f * 0.0], dim=1)

    def forward(self, xt, xw):
        ft = self._feats(xt)
        fw = self._feats(xw)
        return self.head(torch.cat([ft, fw], dim=1))
