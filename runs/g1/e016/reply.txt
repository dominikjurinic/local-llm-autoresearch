    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]

    # Extra training rows: train-split detector boxes that overlap a true box (IoU > 0.5),
    # labelled with that true box's class. Matches the test distribution of stage-1 boxes.
    tr = [d for d in dets if d["split"] == "train"]
    extra = []
    if tr:
        by_img = {}
        for r in crops:
            by_img.setdefault(r["image"], []).append(r)
        for d in tr:
            cands = by_img.get(d["image"], [])
            if not cands:
                continue
            a = np.asarray(cands)
            x1 = np.maximum(d["x1"], a[:, 0]); y1 = np.maximum(d["y1"], a[:, 1])
            x2 = np.minimum(d["x2"], a[:, 2]); y2 = np.minimum(d["y2"], a[:, 3])
            iw = np.clip(x2 - x1, 0, None); ih = np.clip(y2 - y1, 0, None)
            inter = iw * ih
            ad = (d["x2"] - d["x1"]) * (d["y2"] - d["y1"])
            at = (a[:, 2] - a[:, 0]) * (a[:, 3] - a[:, 1])
            iou = inter / np.clip(ad[:, None] + at[:, None] - inter, 1e-9, None)
            i = int(np.argmax(iou))
            if iou[i] > 0.5:
                extra.append({**d, "cls": int(a[i, 4])})
    crops = crops + extra
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
    images = {p: np.asarray(Image.open(os.path.join(args.data, "images", p)).convert("RGB")) for p in needed}
