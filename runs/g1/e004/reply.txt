def precompute_crops(rows, images):
    """Pre-compute both views for every row as uint8 (H,W,C) arrays; no jitter applied."""
    out = []
    for r in rows:
        img = images[r["image"]]
        views = []
        for ctx in (TIGHT_CONTEXT, CONTEXT):
            c = make_crop(img, r, ctx, train=False)  # float32 in [0,1], size IMG_SIZE
            views.append((c * 255.0).astype(np.uint8))
        out.append(views)
    return out

class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, precomputed, images, train):
        self.rows = rows
        self.precomputed = precomputed  # list of [tight_uint8, wide_uint8] per row
        self.images = images
        self.train = train

    def __len__(self):
        return len(self.rows)

    def _jitter_tensor(self, x):
        """Apply random translation + scale to a (3,H,W) float32 tensor in [0,1]."""
        h, w = x.shape[1], x.shape[2]
        # random translation: shift up to 0.1 * side
        tx = int(random.uniform(-0.1, 0.1) * w)
        ty = int(random.uniform(-0.1, 0.1) * h)
        if tx != 0 or ty != 0:
            x = torch.nn.functional.pad(x, (abs(tx), abs(tx), abs(ty), abs(ty)), value=114.0 / 255.0)
            x = x[:, max(0, ty):max(0, ty) + h, max(0, tx):max(0, tx) + w]
        # random scale: 0.85 - 1.2
        sc = random.uniform(0.85, 1.2)
        if abs(sc - 1.0) > 1e-4:
            nh, nw = max(2, int(round(h * sc))), max(2, int(round(w * sc)))
            x = torch.nn.functional.interpolate(x.unsqueeze(0), size=(nh, nw), mode="bilinear", align_corners=False).squeeze(0)
            # center-crop or pad back to (h, w)
            if nh > h or nw > w:
                top = (nh - h) // 2
                left = (nw - w) // 2
                x = x[:, top:top + h, left:left + w]
            else:
                pad_top = (h - nh) // 2
                pad_left = (w - nw) // 2
                x = torch.nn.functional.pad(x, (pad_left, w - nw - pad_left, pad_top, h - nh - pad_top), value=114.0 / 255.0)
        return x

    def __getitem__(self, i):
        r = self.rows[i]
        fh = fv = False
        bright = 1.0
        if self.train:
            fh, fv, bright = random.random() < 0.5, random.random() < 0.5, random.uniform(0.8, 1.2)
        views = []
        for v in self.precomputed[i]:
            x = torch.from_numpy(v.astype(np.float32).transpose(2, 0, 1).copy()) / 255.0
            if self.train:
                x = self._jitter_tensor(x)
            if fh:
                x = x.flip(2)
            if fv:
                x = x.flip(1)
            if self.train:
                x = torch.clamp(x * bright, 0.0, 1.0)
            x = (x - torch.tensor(MEAN).view(3, 1, 1)) / torch.tensor(STD).view(3, 1, 1)
            views.append(x)
        return views[0], views[1], r.get("cls", -1)
