    crops = [r for r in read_csv(os.path.join(args.data, "crops.csv")) if r["res"] in TRAIN_ON]
    dets = read_csv(os.path.join(args.data, "dets.csv"))
    ev = [d for d in dets if d["split"] == "eval"]

    # Train exclusively on the stage-1 detector's own boxes (train split) that overlap a true box:
    # same geometry the model is scored on. Vectorised IoU, per image.
    tr_dets = [d for d in dets if d["split"] == "train"]
    det_rows = []
    for p in {d["image"] for d in tr_dets}:
        tb = [r for r in crops if r["image"] == p]
        if not tb:
            continue
        td = [d for d in tr_dets if d["image"] == p]
        ta = np.array([[r["x1"], r["y1"], r["x2"], r["y2"], r["cls"]] for r in tb], dtype=np.float64)
        da = np.array([[d["x1"], d["y1"], d["x2"], d["y2"]] for d in td], dtype=np.float64)
        ix1 = np.maximum(da[:, :, None, 0], ta[None, None, :, 0])
        iy1 = np.maximum(da[:, :, None, 1], ta[None, None, :, 1])
        ix2 = np.minimum(da[:, :, None, 2], ta[None, None, :, 2])
        iy2 = np.minimum(da[:, :, None, 3], ta[None, None, :, 3])
        inter = np.maximum(0.0, ix2 - ix1) * np.maximum(0.0, iy2 - iy1)
        da_area = (da[:, 2] - da[:, 0]) * (da[:, 3] - da[:, 1])
        ta_area = (ta[:, 2] - ta[:, 0]) * (ta[:, 3] - ta[:, 1])
        iou = inter / np.maximum(1e-9, da_area[:, None, None] + ta_area[None, None, :] - inter)
        best = iou.argmax(axis=2)
        keep = iou.max(axis=2) > 0.5
        for i in np.nonzero(keep)[0]:
            d, j = td[i], tb[best[i]]
            det_rows.append({"image": p, "x1": j[0], "y1": j[1], "x2": j[2], "y2": j[3], "cls": int(j[4])})
    crops = det_rows
    needed = {r["image"] for r in crops} | {d["image"] for d in ev}
    images = {p: np.asarray(Image.open(os.path.join(args.data, "images", p)).convert("RGB")) for p in needed}
    print(f"train {len(crops)} detector boxes, eval {len(ev)} detections, {len(images)} images ({time.time() - T0:.0f}s)", flush=True)
