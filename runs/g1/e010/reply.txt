def make_crop(img, r, context, train, shift_frac=0.0, scale_frac=1.0):
    h, w = img.shape[:2]
    cx, cy = (r["x1"] + r["x2"]) / 2, (r["y1"] + r["y2"]) / 2
    base = max(r["x2"] - r["x1"], r["y2"] - r["y1"])
    side = max(base * context, MIN_CROP)
    if train:  # stage-1 error simulation: shared center shift + scale, IoU with true box ~0.5-1.0
        ang = random.uniform(0, 2 * math.pi)

        cx += m * math.cos(ang)
        cy += m * math.sin(ang)
        side *= random.uniform(scale_frac, 2.0 - scale_frac)
    s = max(2, int(round(side)))
