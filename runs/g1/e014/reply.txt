class BoxDataset(torch.utils.data.Dataset):
    def __init__(self, rows, img_dir, train):
        self.rows, self.img_dir, self.train = rows, img_dir, train
        self._cache = {}

    def __len__(self):
        return len(self.rows)

    def _img(self, name):
        a = self._cache.get(name)
        if a is None:
            a = np.asarray(Image.open(os.path.join(self.img_dir, name)).convert("RGB"))
            if len(self._cache) >= 400:
                self._cache.clear()
            self._cache[name] = a
        return a

    def __getitem__(self, i):
        r = self.rows[i]
        img = self._img(r["image"])
